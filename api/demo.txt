export const useGetMyCompanies = (userId: string) => {
  return useQuery({
    queryKey: ["companies"],
    queryFn: async () => {
      try {
        const response = await axiosInstance.get(`/company/user/${userId}`);
        return response.data.data;
      } catch (error) {
        throw error;
      }
    },
    staleTime: 1000 * 60 * 5,
    enabled: !!userId,
  });
};

export const useSendOtpToEmail = () => {
  return useMutation({
    mutationFn: async ({ email }: { email: string }) => {
      try {
        const response = await axiosInstance.post("/auth/request-code", {
          email,
        });
        return response.data;
      } catch (error) {
        throw error;
      }
    },
  });
};
